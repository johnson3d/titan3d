# Android 原生库: binaries/android/<config>/libCore.Android.so（arm64-v8a / API 28 / c++_static）
# 配置（仓库根）: cmake -S Core.Android -B intermediate/cpp/Core.Android/Debug -G Ninja -DCMAKE_BUILD_TYPE=Debug
#               -DTITAN_CONFIGURATION=Debug -DCMAKE_TOOLCHAIN_FILE=<ndk>/build/cmake/android.toolchain.cmake ...
# 中间产物统一落 intermediate/cpp/<Project>/<Config>，源码目录内构建会被拒绝。
# 源清单来自入库的 generated/TitanSources.cmake；过期时运行 cmake -P Core.Android/RefreshSources.cmake
# 胶水需先在主机生成: cmake -DTITAN_NDK_ROOT=<ndk> -P Core.Android/GenerateCodegen.cmake
cmake_minimum_required(VERSION 3.24)
project(TitanCoreAndroid LANGUAGES NONE)
get_property(_multi GLOBAL PROPERTY GENERATOR_IS_MULTI_CONFIG)
if(_multi)
  message(FATAL_ERROR "Core.Android requires a single-config generator (Ninja)")
endif()
get_filename_component(TITAN_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
include("${TITAN_ROOT}/BuildScript/cmake/TitanPlatformContext.cmake")
titan_platform_context(Android "Core.Android/Core.Android.vcxproj" ANDROID)
include("${TITAN_ROOT}/BuildScript/cmake/TitanSourceList.cmake")
include("${TITAN_ROOT}/BuildScript/cmake/TitanCodegen.cmake")
include("${CMAKE_CURRENT_LIST_DIR}/cmake/TitanAndroidFlags.cmake")
if(NOT CMAKE_BUILD_TYPE STREQUAL TITAN_CONFIGURATION)
  message(FATAL_ERROR "CMAKE_BUILD_TYPE 与 TITAN_CONFIGURATION 必须一致")
endif()
# 中间产物统一落在 intermediate/cpp/Core.Android/<Config>（与 MSBuild 的 intermediate 惯例一致），
# 在 Core.Android 源码目录内 configure 会直接把 CMakeCache/ninja 日志写进仓库树，这里直接拒绝。
include("${TITAN_ROOT}/BuildScript/cmake/TitanBuildLayout.cmake")
titan_require_out_of_source_build("${CMAKE_CURRENT_LIST_DIR}" "${TITAN_ROOT}" "Core.Android" "${TITAN_CONFIGURATION}")
enable_language(C CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Android" OR NOT CMAKE_ANDROID_ARCH_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "只支持 Android arm64-v8a (ABI=${CMAKE_ANDROID_ARCH_ABI})")
endif()
# project(LANGUAGES NONE) 下 NDK 工具链不会把 API 回填到 CMAKE_SYSTEM_VERSION（恒为 1），
# 而它已按 ANDROID_PLATFORM 选中 sysroot/usr/lib/<triple>/28；此处直接校验输入契约。
if(NOT ANDROID_PLATFORM STREQUAL "28" AND NOT ANDROID_PLATFORM STREQUAL "android-28")
  message(FATAL_ERROR "只支持 API 28 (ANDROID_PLATFORM=${ANDROID_PLATFORM})")
endif()
if(NOT ANDROID_STL STREQUAL "c++_static")
  message(FATAL_ERROR "只支持 c++_static (ANDROID_STL=${ANDROID_STL})")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 11)
# 生成清单必须先与平台工程一致；过期时不让增量构建带着旧源继续。
execute_process(COMMAND "${CMAKE_COMMAND}" -DTITAN_CHECK=ON
  -P "${CMAKE_CURRENT_LIST_DIR}/RefreshSources.cmake" COMMAND_ERROR_IS_FATAL ANY)
titan_sources_fingerprint(_fp "${TITAN_ROOT}" ${TITAN_SOURCES_INPUTS})
if(NOT _fp STREQUAL TITAN_SOURCES_FINGERPRINT)
  message(FATAL_ERROR "[titan-sources] Android 清单已过期；运行 RefreshSources.cmake")
endif()
foreach(_in IN LISTS TITAN_SOURCES_INPUTS)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${TITAN_ROOT}/${_in}")
endforeach()
set(_norm "${CMAKE_CURRENT_BINARY_DIR}/codegen_norm")
titan_prepare_codegen(_glue "${TITAN_ROOT}" "${_norm}")
list(TRANSFORM TITAN_MODULE_SOURCES PREPEND "${TITAN_ROOT}/" OUTPUT_VARIABLE _sources)
list(TRANSFORM TITAN_ANDROID_INCLUDES PREPEND "${TITAN_ROOT}/" OUTPUT_VARIABLE _includes)
add_library(CoreAndroid SHARED ${_sources} ${_glue} "${CMAKE_CURRENT_LIST_DIR}/Core_Android.cpp")
target_compile_definitions(CoreAndroid PRIVATE ${TITAN_ANDROID_DEFINES})
target_compile_options(CoreAndroid PRIVATE -fPIC -fexceptions -frtti -w -ferror-limit=50)
target_include_directories(CoreAndroid PRIVATE "${_norm}" "${TITAN_ROOT}" ${_includes} "${CMAKE_CURRENT_LIST_DIR}")
if("3rd/native/zlib/zlib.vcxitems" IN_LIST TITAN_ANDROID_MODULES)
  target_sources(CoreAndroid PRIVATE "${TITAN_ROOT}/3rd/native/zlib/inflate.c" "${TITAN_ROOT}/3rd/native/zlib/uncompr.c")
endif()
if("3rd/native/FreeType/FreeType.vcxitems" IN_LIST TITAN_ANDROID_MODULES)
  target_sources(CoreAndroid PRIVATE "${TITAN_ROOT}/3rd/native/FreeType/src/base/ftsystem.c")
endif()
foreach(_entry IN LISTS TITAN_ANDROID_SOURCE_DEFINES)
  string(REPLACE "|" ";" _parts "${_entry}")
  list(GET _parts 0 _source)
  list(GET _parts 1 _config)
  list(GET _parts 2 _define)
  set_property(SOURCE "${TITAN_ROOT}/${_source}" APPEND PROPERTY COMPILE_DEFINITIONS "$<$<CONFIG:${_config}>:${_define}>")
endforeach()
if("3rd/native/LeanCLR/LeanCLR.vcxitems" IN_LIST TITAN_ANDROID_MODULES)
  set(BUILD_SHARED_LEANCLR OFF CACHE BOOL "LeanCLR 静态库" FORCE)
  if(CMAKE_BUILD_TYPE STREQUAL "Debug")
    set(LEANCLR_GC_DEBUG ON CACHE BOOL "LeanCLR GC 检查" FORCE)
  else()
    set(LEANCLR_GC_DEBUG OFF CACHE BOOL "LeanCLR GC 检查" FORCE)
  endif()
  add_subdirectory("${TITAN_ROOT}/3rd/native/LeanCLR/runtime" "${CMAKE_CURRENT_BINARY_DIR}/leanclr")
  set_target_properties(leanclr PROPERTIES POSITION_INDEPENDENT_CODE ON)
  target_link_libraries(CoreAndroid PRIVATE leanclr)
endif()
# PhysX 只提供 release 静态库产物，Debug/Release 都链接它，不重编 debug。
set(_px_dir "${TITAN_ROOT}/3rd/native/PhysX-4.1/physx/lib/android/arm64-v8a/release")
set(_px_targets)
foreach(_lib PhysXCharacterKinematic PhysXVehicle PhysXExtensions PhysX PhysXPvdSDK PhysXCooking PhysXCommon PhysXFoundation)
  add_library(px_${_lib} STATIC IMPORTED)
  set_target_properties(px_${_lib} PROPERTIES IMPORTED_LOCATION "${_px_dir}/lib${_lib}_static_64.a")
  list(APPEND _px_targets px_${_lib})
endforeach()
# 引擎直接调用 Vulkan 1.3 核心入口（vkCmdBeginRendering / vkCmdPipelineBarrier2 / vkGetBufferDeviceAddress…），
# 而 ANDROID_PLATFORM=28 选中的 NDK stub 只导出到 Vulkan 1.1；API 33 才是首个包含这些符号的 stub。
# 这里只影响链接期符号解析，NEEDED 仍是 libvulkan.so；运行期要求设备的 libvulkan.so 提供 Vulkan 1.3。
set(TITAN_ANDROID_VULKAN_STUB_LEVEL 33 CACHE STRING "链接 libvulkan.so stub 使用的 API 级别")
set(_vk_stub "${CMAKE_SYSROOT}/usr/lib/aarch64-linux-android/${TITAN_ANDROID_VULKAN_STUB_LEVEL}/libvulkan.so")
if(NOT EXISTS "${_vk_stub}")
  message(FATAL_ERROR "找不到 libvulkan.so stub: ${_vk_stub}")
endif()
target_link_libraries(CoreAndroid PRIVATE "-Wl,--start-group" ${_px_targets} "-Wl,--end-group" log android "${_vk_stub}" m dl)
target_link_options(CoreAndroid PRIVATE "LINKER:--no-undefined"
  "LINKER:--version-script=${CMAKE_CURRENT_LIST_DIR}/exported.map" "$<$<CONFIG:Release>:-s>")
set_property(TARGET CoreAndroid APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/exported.map")
set_target_properties(CoreAndroid PROPERTIES OUTPUT_NAME "Core.Android"
  LIBRARY_OUTPUT_DIRECTORY "${TITAN_ROOT}/binaries/android/$<LOWER_CASE:$<CONFIG>>")
